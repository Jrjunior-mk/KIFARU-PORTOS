cmake_minimum_required(VERSION 3.16)
project(wim_dsp_host_test C)
set(CMAKE_C_STANDARD 11)
add_executable(test_wim_dsp test_wim_dsp.c ../App/wim_dsp.c ../App/frames.c)
target_include_directories(test_wim_dsp PRIVATE ../App)
target_compile_definitions(test_wim_dsp PRIVATE HOST_TEST=1)
if(NOT MSVC)
  target_link_libraries(test_wim_dsp m)
  target_compile_options(test_wim_dsp PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME wim_dsp COMMAND test_wim_dsp)
